/* =====================================================================
   Measure : Star Rating Card
   Group   : _Service Quality \ 4. Report-Only Visuals
   Model   : Airline Passenger Satisfaction
   Table   : 'Airline Quality Ratings'
   Returns : HTML/SVG string (rendered by an HTML-capable visual)
   Depends : [Average Rating]  (defined in airline_service_quality_measures.md)
   Version : 1.0
   Date    : 2026-10-24

   Renders a 5-star SVG with a partial-fill (via clipPath) that shows the
   current [Average Rating] out of 5. Pure front-end: no JavaScript.

   Requires : an HTML-rendering visual
              (e.g. "HTML Content" by Daniel Marsh-Patrick).

   Do NOT   : use in aggregations, conditional formatting, axis bindings,
              tooltips, or anywhere a numeric return value is expected.
              The measure returns a string.

   Caveats  : see companion doc `star_rating_card.md`
   ===================================================================== */

Star Rating Card =
VAR Rating    = COALESCE ( [Average Rating], 0 )
VAR MaxRating = 5
VAR ClipWidth = DIVIDE ( MIN ( Rating, MaxRating ), MaxRating ) * 120
VAR ClipText  = SUBSTITUTE ( FORMAT ( ClipWidth, "0.00" ), ",", "." )
VAR StarPath  = "M12 2l3.09 6.26L22 9.27l-5 4.87 1.18 6.88L12 17.77l-6.18 3.25L7 14.14 2 9.27l6.91-1.01z"

RETURN
"<div style='text-align:center;padding:6px;'>
  <svg viewBox='0 0 120 24' width='150'>
    <defs>
      <g id='stars'>
        <path d='" & StarPath & "'/>
        <path transform='translate(24 0)' d='" & StarPath & "'/>
        <path transform='translate(48 0)' d='" & StarPath & "'/>
        <path transform='translate(72 0)' d='" & StarPath & "'/>
        <path transform='translate(96 0)' d='" & StarPath & "'/>
      </g>
      <clipPath id='fillClip'><rect x='0' y='0' width='" & ClipText & "' height='24'/></clipPath>
    </defs>
    <use href='#stars' fill='#d9d9d9'/>
    <g clip-path='url(#fillClip)'><use href='#stars' fill='#FFB400'/></g>
  </svg>
</div>"
